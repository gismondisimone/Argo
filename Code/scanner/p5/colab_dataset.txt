%%writefile compile.py
import os
import numpy as np
from PIL import Image
from hailo_sdk_client import ClientRunner

# Path to the exported ONNX model
onnx_path = '/content/runs/classify/train/weights/best.onnx'

if not os.path.exists(onnx_path):
    # Fallback check for Google Drive path
    onnx_path = '/content/drive/MyDrive/Argo/CroatiArgo.onnx'

print(f"ONNX path: {onnx_path}")

if not os.path.exists(onnx_path):
    print("⚠️File ONNX not found")
else:
    # Function to create calibration dataset in (N, H, W, C) format
    def get_calibration_data(val_dir, num_samples=64, img_size=(224, 224)):
        image_paths = []
        for root, _, files in os.walk(val_dir):
            for f in files:
                if f.lower().endswith(('.png', '.jpg', '.jpeg')):
                    image_paths.append(os.path.join(root, f))
                    if len(image_paths) >= num_samples:
                        break
            if len(image_paths) >= num_samples:
                break

        images = []
        for p in image_paths:
            img = Image.open(p).convert('RGB').resize(img_size)
            arr = np.array(img, dtype=np.float32) / 255.0  # Keep in (H, W, C) format (224, 224, 3)
            images.append(arr)

        if not images:
            print("⚠️ Validation images not found, generating synthetic calibration data...")
            return np.random.uniform(0.0, 1.0, (num_samples, img_size[0], img_size[1], 3)).astype(np.float32)

        return np.array(images, dtype=np.float32)

    calib_dataset = get_calibration_data('/content/Argo-2/val')

    # Pass target hardware architecture to ClientRunner
    runner = ClientRunner(hw_arch='hailo8l')

    runner.translate_onnx_model(
        onnx_path,
        'argo_model',
        start_node_names=['images'],
        end_node_names=['/model.9/Softmax']
    )

    # Pass calibration data to optimization
    runner.optimize(calib_dataset)
    hef = runner.compile()

    with open('argo_materiali.hef', 'wb') as f:
        f.write(hef)

    print("All done, download 'argo_materiali.hef' from the left folder")
